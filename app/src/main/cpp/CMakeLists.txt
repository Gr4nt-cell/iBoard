cmake_minimum_required(VERSION 3.22.1)
project(iboard_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_AARCH64 OFF CACHE BOOL "" FORCE)
# Build against the pinned source; no network operation occurs inside the app.
add_subdirectory(whisper.cpp)
# IME inference must stay optimized even when the Android application is a debug build.
# Optimizing just the JNI shim leaves the actual transformer/matrix kernels at -O0.
foreach(engine_target whisper ggml ggml-base ggml-cpu)
    if(TARGET ${engine_target})
        target_compile_options(${engine_target} PRIVATE -O3 -DNDEBUG)
    endif()
endforeach()
add_library(iboard_whisper SHARED whisper_jni.cpp)
target_compile_options(iboard_whisper PRIVATE -O3 -fvisibility=hidden)
target_link_options(iboard_whisper PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(iboard_whisper PRIVATE whisper android log)
